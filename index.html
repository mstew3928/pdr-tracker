package com.example.offlinenav import android.content.Context import android.hardware.Sensor import android.hardware.SensorEvent import android.hardware.SensorEventListener import android.hardware.SensorManager import kotlin.math.cos import kotlin.math.sin data class Position3D( val xMeters: Float, val yMeters: Float, val zAltitudeMeters: Float, val headingDegrees: Float, val stepCount: Int ) class PdrTracker( context: Context, private val strideLengthMeters: Float = 0.72f, // Average human stride; can be made dynamic private val onPositionUpdated: (Position3D) -> Unit ) : SensorEventListener { private val sensorManager = context.getSystemService(Context.SENSOR_SERVICE) as SensorManager private val stepSensor: Sensor? = sensorManager.getDefaultSensor(Sensor.TYPE_STEP_DETECTOR) private val rotationSensor: Sensor? = sensorManager.getDefaultSensor(Sensor.TYPE_ROTATION_VECTOR) private val pressureSensor: Sensor? = sensorManager.getDefaultSensor(Sensor.TYPE_PRESSURE) private var posX = 0f private var posY = 0f private var posZ = 0f private var baselinePressure: Float? = null private var currentHeadingRad = 0f private var totalSteps = 0 private val rotationMatrix = FloatArray(9) private val orientationAngles = FloatArray(3) fun start() { stepSensor?.let { sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_FASTEST) } rotationSensor?.let { sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_GAME) } pressureSensor?.let { sensorManager.registerListener(this, it, SensorManager.SENSOR_DELAY_UI) } } fun stop() { sensorManager.unregisterListener(this) } fun resetOrigin() { posX = 0f posY = 0f posZ = 0f totalSteps = 0 baselinePressure = null emitUpdate() } override fun onSensorChanged(event: SensorEvent) { when (event.sensor.type) { // 1. Fused 3D Orientation (Gyro + Accel + Magnetometer) Sensor.TYPE_ROTATION_VECTOR -> { SensorManager.getRotationMatrixFromVector(rotationMatrix, event.values) SensorManager.getOrientation(rotationMatrix, orientationAngles) // orientationAngles[0] is azimuth (yaw) in radians (-PI to +PI) currentHeadingRad = orientationAngles[0] emitUpdate() } // 2. Hardware Step Trigger -> Advance (X, Y) along current heading Sensor.TYPE_STEP_DETECTOR -> { if (event.values[0] == 1.0f) { totalSteps++ posX += strideLengthMeters * sin(currentHeadingRad) posY += strideLengthMeters * cos(currentHeadingRad) emitUpdate() } } // 3. Barometer -> Relative Z-Axis Elevation (Underground / Multi-floor) Sensor.TYPE_PRESSURE -> { val currentPressure = event.values[0] if (baselinePressure == null) { baselinePressure = currentPressure } posZ = SensorManager.getAltitude(baselinePressure!!, currentPressure) emitUpdate() } } } override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {} private fun emitUpdate() { val headingDeg = ((Math.toDegrees(currentHeadingRad.toDouble()).toFloat() + 360f) % 360f) onPositionUpdated(Position3D(posX, posY, posZ, headingDeg, totalSteps)) } }
